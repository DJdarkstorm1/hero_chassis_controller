namespace simple_chassis_controller {//插件的命名空间
//插件类被加载到manager时调用的初始化函数
bool SimpleChassisController::init(
    //力矩控制关节硬件接口
    hardware_interface::EffortJointInterface *effort_joint_interface,
    //根命名空间，用于获取参数等
    ros::NodeHandle &root_nh, 
    //控制器命名空间
    ros::NodeHandle &controller_nh) {
    //获取关节句柄,句柄名称根据URDF文件中的关节名称获取
  front_left_joint_ =
      effort_joint_interface->getHandle("left_front_wheel_joint");
  front_right_joint_ =
      effort_joint_interface->getHandle("right_front_wheel_joint");
  back_left_joint_ = 
  	effort_joint_interface->getHandle("left_back_wheel_joint");
  back_right_joint_ =
      effort_joint_interface->getHandle("right_back_wheel_joint");

  return true;
}
//插件运行时不断调用的update函数
void SimpleChassisController::update(const ros::Time &time, const ros::Duration &period) {
    //由于是力矩关节，得发送力矩数据
  double tau = 0.2;  // torque
  // NOTE: DON'T COPY THESE NAIVE TESTING CODES !!! USE INVERSE KINEMATICS !!!
  // NOTE: DON'T COPY THESE NAIVE TESTING CODES !!! USE INVERSE KINEMATICS !!!
  // NOTE: DON'T COPY THESE NAIVE TESTING CODES !!! USE INVERSE KINEMATICS !!!
  //状态机
  static double cmd_[6][4] = {{tau, tau, tau, tau}, //  forward
                              {-2 * tau, -2 * tau, -2 * tau, -2 * tau}, //  backward
                              {-tau, tau, tau, -tau}, //  left
                              {2 * tau, -2 * tau, -2 * tau, 2 * tau}, //  right
                              {2 * tau, -2 * tau, 2 * tau, -2 * tau}, //  clockwise
                              {-tau, tau, -tau, tau}};  //  counterclockwise
  if ((time - last_change_).toSec() > 2) {
    state_ = (state_ + 1) % 6;
    last_change_ = time;
  }
  //发布关节命令（模板类forward_command_controller的成员函数）
  front_left_joint_.setCommand(cmd_[state_][0]);
  front_right_joint_.setCommand(cmd_[state_][1]);
  back_left_joint_.setCommand(cmd_[state_][2]);
  back_right_joint_.setCommand(cmd_[state_][3]);
}
//导出类插件：允许类动态加载，实际上是将类注册到ROS构建/打包到系统中
PLUGINLIB_EXPORT_CLASS(
    simple_chassis_controller::SimpleChassisController,//子类
    controller_interface::ControllerBase//父类
)
}

2. cmdVelCallback()函数

   ```
   void HeroChassisController::cmdVelCallback(const geometry_msgs::Twist &msg)
   //逆运动学解算
       double vel_x = msg.linear.x;
       double vel_y = msg.linear.y;
       double vel_rad = msg.angular.z;
   
       double fl_vel = (vel_x - vel_y -(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
       double fr_vel = (vel_x + vel_y +(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
       double bl_vel = (vel_x + vel_y -(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
       double br_vel = (vel_x - vel_y +(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
   
   ```

2. 速度消息回调函数内进行速度变换

   ```c++
   //底盘/全局
   if (global_mode_) {
       try {//全局模式
           // 1. lookupTransform()函数查询 odom → base_link 的变换
           //    参数顺序："target_frame子坐标系", "source_frame父坐标系", time时间传0即可，传入当下的时间会出错，因为buffer的数据有延迟
           //    返回的 transform 表示把 source 中的数据变换到 target
           geometry_msgs::TransformStamped transform_stamped =
               tf_buffer_->lookupTransform(
                   base_frame_,   // 子坐标系
                   odom_frame_,   // 父坐标系
                   ros::Time(0),
                   ros::Duration(0.1));  // 超时 100ms，避免阻塞太久
   
           // 2. 速度是自由向量，只需取出旋转部分，转成 tf2::Quaternion
           tf2::Quaternion q;
           tf2::fromMsg(transform_stamped.transform.rotation, q);
   
           // 3. 用四元数旋转线速度向量（速度是自由向量，只旋转不平移）
           tf2::Vector3 v_world(msg.linear.x, msg.linear.y, 0.0);
           tf2::Vector3 v_body = tf2::quatRotate(q, v_world);
   
           // 4. 写回 Twist（angular.z 在平面运动中不变）
           vel_x = v_body.x();
           vel_y = v_body.y();
   
       } catch (const tf2::TransformException& ex) {
           ROS_WARN_THROTTLE(1.0,
               "TF2 lookup %s -> %s failed: %s, using original command",
               odom_frame_.c_str(), base_frame_.c_str(), ex.what());
       }
   }
   else
   {	//底盘模式
       vel_x = msg.linear.x;
       vel_y = msg.linear.y;
   }
   vel_rad = msg.angular.z;//转向不变
   ```

2. 加速度限制函数

   **参考：[yocs_velocity_smoother](https://github.com/yujinrobot/yujin_ocs/tree/devel)**

   ```c++
   void HeroChassisController::applyVelocityLimits(geometry_msgs::Twist& cmd,const ros::Duration& period)
   {
       //记录时间间隔
       const double dt = period.toSec();
       if (dt <= 0.0) return;
   
       //线速度:vx,vy合成为合成增量
       const double vx_cur = last_smoothed_cmd_.linear.x;
       const double vy_cur = last_smoothed_cmd_.linear.y;
       const double vx_inc  = cmd.linear.x - vx_cur;//计算增量
       const double vy_inc  = cmd.linear.y - vy_cur;
   
       const double linear_inc_mag = std::hypot(vx_inc, vy_inc);//计算目标直线增量的模
       const double linear_cur_mag = std::hypot(vx_cur, vy_cur);//计算当前速度向量的模
       const double linear_tgt_mag = std::hypot(cmd.linear.x, cmd.linear.y);//目标速度向量的模
   
       //判断加/减速：看合成速度幅值是变大还是变小
       const bool linear_acc = linear_tgt_mag >= linear_cur_mag;
       const double max_linear_inc_x = (linear_acc ? max_lin_acc_x_ : max_lin_dec_x_)*dt;//计算最大速度增量
       const double max_linear_inc_y = (linear_acc ? max_lin_acc_y_ : max_lin_dec_y_)*dt;
       const double max_linear_inc = std::hypot(max_linear_inc_x, max_linear_inc_y);//最大增量的模
   
       //角速度：(同理)
       const double wz_cur = last_smoothed_cmd_.angular.z;
       const double wz_inc  = cmd.angular.z - wz_cur;
       const bool wz_acc = (std::abs(cmd.angular.z) >= std::abs(wz_cur));
       const double max_wz_inc = (wz_acc ? max_ang_acc_z_ : max_ang_dec_z_)*dt;
   
       //联合约束，保证前进方向
       double allowed_linear_inc = max_linear_inc;
       double allowed_wz_inc = max_wz_inc;
   
       if (linear_inc_mag > 1e-9 && std::abs(wz_inc)>1e-9)
       {
           const double MA = std::hypot(linear_inc_mag,std::abs(wz_inc));
           const double MB = std::hypot(max_linear_inc,std::abs(wz_inc));
           const double Av = linear_inc_mag/MA;
           const double Aw = std::abs(wz_inc)/MA;
           const double Bv = max_linear_inc/MB;
           const double Bw = max_wz_inc/MB;
           const double theta = std::atan2(Bw,Bv) - std::atan2(Aw,Bv);
   
           if (theta > 0.0)
           {
               //角速度方向更陡，压缩合成直线增量，保持 vx:vy 比例
               allowed_linear_inc = max_wz_inc*linear_inc_mag/std::abs(wz_inc);
           }
           else
           {
               //直线方向更陡，压缩角速度增量
               allowed_wz_inc = max_linear_inc*std::abs(wz_inc)/linear_inc_mag;
           }
       }
   
       double linear_scale = 1.0;
       if (linear_inc_mag > 1e-9 && linear_inc_mag > allowed_linear_inc)
       {
           linear_scale = allowed_linear_inc/linear_inc_mag;
       }
       cmd.linear.x = vx_cur + vx_inc * linear_scale;
       cmd.linear.y = vy_cur + vy_inc * linear_scale;
   
       if (std::abs(wz_inc) > allowed_wz_inc ) {
           cmd.angular.z = wz_cur + (wz_inc > 0 ? 1.0 : -1.0 ) * allowed_wz_inc;
       }
       else {
           cmd.angular.z = wz_cur + wz_inc ;
       }
   
       last_smoothed_cmd_ = cmd;//更新速度平滑后的状态
   
   }
   ```

       double totol_cmd_torque = 0.0;

       for (int i=0; i<commanded_efforts_.size(); i++) {
           out_Power_ += std::abs(commanded_efforts_[i] * joints_[i].getVelocity());
           totol_cmd_torque += commanded_efforts_[i]/19 * commanded_efforts_[i]/19;
           totol_w += joints_[i].getVelocity()*19 *joints_[i].getVelocity()*19;
       }
       in_Power_ = out_Power_ + k1 * totol_cmd_torque + k2 * totol_w;
   	power_msg_.ori_power = in_Power_;
       if (in_Power_ > max_Power_) {
           double A = k1 * totol_cmd_torque;
           double B = out_Power_;
           double C = k2 * totol_w - max_Power_;
   
           // 计算判别式 Δ = B^2 - 4AC
           double delta = B * B - 4.0 * A * C;
   
           // 防止分母为 0 (A 为 0 说明所有力矩指令为 0)，且保证根号下非负
           if (std::abs(A) > 1e-6 && delta >= 0.0) {
               scale_k_ = (-B + std::sqrt(delta)) / (2.0 * A);
   
               // 防止浮点数误差导致 k 大于 1 或小于 0
               if (scale_k_ > 1.0) scale_k_ = 1.0;
               if (scale_k_ < 0.0) scale_k_ = 0.0;
           } else {
               // 如果无解，直接切断动力
               scale_k_ = 0.0;
           }
   
           for (double & commanded_effort : commanded_efforts_) {
               commanded_effort *= scale_k_;
           }
       }
   
       out_Power_ = 0.0;
       totol_cmd_torque = 0.0;
       totol_w = 0.0;
       for (int i=0; i<commanded_efforts_.size(); i++) {
           out_Power_ += std::abs(commanded_efforts_[i] * joints_[i].getVelocity());
           totol_cmd_torque += commanded_efforts_[i]/19 * commanded_efforts_[i]/19;
           totol_w += joints_[i].getVelocity()*19 *joints_[i].getVelocity()*19;
       }
       in_Power_ = out_Power_ + k1 * totol_cmd_torque + k2 * totol_w;
   
       power_msg_.limit_power = in_Power_;
       power_msg_.max_power = max_Power_;
       power_pub_.publish(power_msg_);
   }

### 4.launch文件标签

https://blog.csdn.net/x_r_su/article/details/54952880

1. 根标签launch
   launch标签是所有 launch 文件的根标签，充当其他标签的容器。
   属性：

```
deprecated = “弃用声明” 告知用户当前 launch 文件已经弃用。运行时，会有红色的警告信息，说此文件已经过时或者弃用。
```
2. node标签
   node标签用于指定 ROS 节点，是最常见的标签，需要注意的是: roslaunch 命令不能保证按照 node 的声明顺序来启动节点(节点的启动是多进程的)
   属性：

```
pkg=“包名” 节点所属的包
type=“nodeType” 节点类型(与之相同名称的可执行文件)
name=“nodeName” 节点名称(在 ROS 网络拓扑中节点的名称)
args=“xxx xxx xxx” (可选) 将参数传递给节点
machine=“机器名” 在指定机器上启动节点
respawn=“true | false” (可选) 如果节点退出，是否自动重启
respawn_delay=" N" (可选) 如果 respawn 为 true, 那么延迟 N 秒后启动节点
required=“true | false” (可选) 该节点是否必须，如果为 true,那么如果该节点退出，将杀死整个 roslaunch
ns=“xxx” (可选) 在指定命名空间 xxx 中启动节点
clear_params=“true | false” (可选) 在启动前，删除节点的私有空间的所有参数
output=“log | screen” (可选) 日志发送目标，可以设置为 log 日志文件，或 screen 屏幕,默认是 log

子级标签

env 环境变量设置
remap 重映射节点名称
rosparam 参数设置
param 参数设置
```
3. include标签
   include标签用于将另一个 xml 格式的 launch 文件导入到当前文件

```
file=“$(find 包名)/xxx/xxx.launch” 要包含的文件路径
ns=“xxx” (可选) 在指定命名空间导入文件

<launch>
    <include file="$(find launch所在项目名)/launch/***.launch"/>
</launch>
```
4. remap 标签
   该标签用于话题的重命名。有时候某些话题需要与其他话题进行同步，则需要该命令进行重新映射。

```
from=“xxx” 原始话题名称
to=“yyy” 目标名称
```
