2. `SimpleChassisController` 的类成员意义

```c++
namespace simple_chassis_controller {//插件的命名空间
//插件类被加载到manager时调用的初始化函数
bool SimpleChassisController::init(
    //力矩控制关节硬件接口
    hardware_interface::EffortJointInterface *effort_joint_interface,
    //根命名空间，用于获取参数等
    ros::NodeHandle &root_nh, 
    //控制器命名空间
    ros::NodeHandle &controller_nh) {
    //获取关节句柄,句柄名称根据URDF文件中的关节名称获取
  front_left_joint_ =
      effort_joint_interface->getHandle("left_front_wheel_joint");
  front_right_joint_ =
      effort_joint_interface->getHandle("right_front_wheel_joint");
  back_left_joint_ = 
  	effort_joint_interface->getHandle("left_back_wheel_joint");
  back_right_joint_ =
      effort_joint_interface->getHandle("right_back_wheel_joint");

  return true;
}
//插件运行时不断调用的update函数
void SimpleChassisController::update(const ros::Time &time, const ros::Duration &period) {
    //由于是力矩关节，得发送力矩数据
  double tau = 0.2;  // torque
  // NOTE: DON'T COPY THESE NAIVE TESTING CODES !!! USE INVERSE KINEMATICS !!!
  // NOTE: DON'T COPY THESE NAIVE TESTING CODES !!! USE INVERSE KINEMATICS !!!
  // NOTE: DON'T COPY THESE NAIVE TESTING CODES !!! USE INVERSE KINEMATICS !!!
  //状态机
  static double cmd_[6][4] = {{tau, tau, tau, tau}, //  forward
                              {-2 * tau, -2 * tau, -2 * tau, -2 * tau}, //  backward
                              {-tau, tau, tau, -tau}, //  left
                              {2 * tau, -2 * tau, -2 * tau, 2 * tau}, //  right
                              {2 * tau, -2 * tau, 2 * tau, -2 * tau}, //  clockwise
                              {-tau, tau, -tau, tau}};  //  counterclockwise
  if ((time - last_change_).toSec() > 2) {
    state_ = (state_ + 1) % 6;
    last_change_ = time;
  }
  //发布关节命令（模板类forward_command_controller的成员函数）
  front_left_joint_.setCommand(cmd_[state_][0]);
  front_right_joint_.setCommand(cmd_[state_][1]);
  back_left_joint_.setCommand(cmd_[state_][2]);
  back_right_joint_.setCommand(cmd_[state_][3]);
}
//导出类插件：允许类动态加载，实际上是将类注册到ROS构建/打包到系统中
PLUGINLIB_EXPORT_CLASS(
    simple_chassis_controller::SimpleChassisController,//子类
    controller_interface::ControllerBase//父类
)
}	
```

#### 6. 使用逆运动学计算各个轮子的期望速度

1. 头文件

   ```
   #include <geometry_msgs/Twist.h>
   void cmdVelCallback(const geometry_msgs::Twist &msg);
   ros::Subscriber sub_command_;
   //从参数服务器获取基本参数，以便后续解算
   double wheel_base_;//前后轴距
   double wheel_radius_;//轮子半径
   double wheel_track_;//左右轮距
   ```

2. update()函数

   ```
   void HeroChassisController::cmdVelCallback(const geometry_msgs::Twist &msg)
   //逆运动学解算
       double vel_x = msg.linear.x;
       double vel_y = msg.linear.y;
       double vel_rad = msg.angular.z;
   
       double fl_vel = (vel_x - vel_y -(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
       double fr_vel = (vel_x + vel_y +(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
       double bl_vel = (vel_x + vel_y -(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
       double br_vel = (vel_x - vel_y +(wheel_base_+wheel_track_)*vel_rad)/wheel_radius_;
   
   ```

# 一些笔记

## 1.创建插件

1. 准备

   + 导入依赖

   ```
   find_package(pluginlib)
   include_directories(
     include
     ${catkin_INCLUDE_DIRS}
   )
   add_library(hero_chassis_controller
       src/hero_chassis_controller.cpp
   )
   ```

   

2. 创建基类(此项目不必)

3. 创建插件类

   ```
   #include "hero_chassis_controller/hero_chassis_controller.h"
   #include <pluginlib/class_list_macros.hpp>
   ```

   

4. 注册插件

   ```
   PLUGINLIB_EXPORT_CLASS(
   	simple_chassis_controller::SimpleChassisController,//子类
   	controller_interface::ControllerBase//父类
   	)
   ```

   

5. 构建插件库

   ```
   catkin build
   # 会在 include/包名 文件夹下生成.h文件
   # devel/lib 下生成libxxx.so文件
   ```

   

6. 使插件可用于ROS工具链：

   + 配置hero_chassis_controller.xml

     ```
     <library path="lib/libhero_chassis_controller">
     
         <class name="hero_chassis_controller/HeroChassisController"
                type="hero_chassis_controller::HeroChassisController"
                base_class_type="controller_interface::ControllerBase">
             <description>
                 The HeroChassisController is RoboMaster standard robot Chassis controller. It expects a EffortJointInterface
                 type of hardware interface.
             </description>
         </class>
     
     </library>
     ```

     

   + 导出插件

     ```
     <export>
         <!-- Other tools can request additional information be placed here -->
         <controller_interface plugin="${prefix}/hero_chassis_controller.xml"/>
       </export>
     ```
